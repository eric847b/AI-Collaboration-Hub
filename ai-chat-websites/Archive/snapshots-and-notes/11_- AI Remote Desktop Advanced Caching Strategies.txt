(function() {
    'use strict';

    function cacheInvalidation() {
        console.log("Cache invalidation strategy...");
        const cache = new Map();
        function setCache(key, value, ttl) {
            const expiry = Date.now() + ttl;
            cache.set(key, { value, expiry });
        }
        function getCache(key) {
            const cached = cache.get(key);
            if (cached && cached.expiry > Date.now()) {
                return cached.value;
            } else {
                cache.delete(key);
                return null;
            }
        }
        // Example usage
        setCache('exampleKey', 'exampleValue', 60000); // Cache for 60 seconds
        console.log(getCache('exampleKey')); // Should print 'exampleValue'
    }
